  claim_tiers:

    interpretation:
      definition: >
        A meaning, explanation, or causal reading that organizes information but does
        not assert measured performance or validation.
      allowed_sources:
        - Lexicon.yaml
        - CE-OS.yaml
        - Interface.yaml
        - user_notes
      required_evidence:
        - source_context
        - authority_lane
      prohibited_escalation:
        - validated_claim
        - operational_readiness
        - safety_claim

    hypothesis:
      definition: >
        A testable proposed relationship, mechanism, or expected behavior that requires
        measurement before promotion.
      allowed_sources:
        - Physics.yaml
        - Interface.yaml
        - Engine.yaml
        - interpretation_records
      required_evidence:
        - stated_variables
        - expected_interaction_path
        - unresolved_measurements
        - test_condition_candidate
      prohibited_escalation:
        - proof_claim
        - validated_performance
        - broad_operating_envelope

    modeled_result:
      definition: >
        Output from an Engine operator, format, simulation, or structured calculation.
      allowed_sources:
        - Engine.yaml
        - measurement_records
      required_evidence:
        - input_variables
        - assumptions
        - calculation_method
        - units
        - boundary_conditions
        - unresolved_losses
      prohibited_escalation:
        - measurement_equivalence
        - validation_equivalence
        - safety_equivalence

    observed_behavior:
      definition: >
        Behavior seen or recorded but not yet fully measured, repeated, or bounded.
      allowed_sources:
        - test_notes
        - inspection_records
        - field_observations
        - images_or_video_with_context
      required_evidence:
        - observation_context
        - date_or_test_condition
        - object_identity
        - observed_state_change
      prohibited_escalation:
        - repeatability_claim
        - validated_performance
        - long_term_durability

    measured_record:
      definition: >
        Recorded measurement with method, unit, context, and object or system identity.
      allowed_sources:
        - measurement_logs
        - instrument_readings
        - structured_test_records
      required_evidence:
        - value
        - unit
        - method
        - condition
        - object_identity
        - timestamp_or_trial_id
      prohibited_escalation:
        - validation_without_repeatability
        - broad_claim_without_envelope

    bounded_result:
      definition: >
        Repeated or constrained measurement record defining a limited condition envelope.
      allowed_sources:
        - repeated_measurements
        - controlled_tests
        - bounded_operator_outputs
      required_evidence:
        - repeat_count
        - boundary_conditions
        - variance_or_uncertainty
        - anomaly_records
        - negative_results_if_any
      prohibited_escalation:
        - extrapolation_beyond_tested_conditions
        - universal_claim
        - operational_readiness

    validated_claim:
      definition: >
        Claim promoted only after evidence review confirms repeatability, scope,
        boundary conditions, anomaly handling, and explicit admissibility.
      allowed_sources:
        - governance_review
        - validation_package
      required_evidence:
        - claim_text
        - supporting_measurements
        - tested_envelope
        - repeatability_threshold
        - anomaly_log
        - review_decision
      prohibited_escalation:
        - extrapolation_beyond_scope
        - implied_commercial_readiness
        - implied_safety_without_specific_review

  evidence_states:

    none:
      definition: no evidence record attached
      permitted_claim_tiers:
        - interpretation
      required_action: add_source_context_or_keep_as_note

    candidate:
      definition: structured idea or model exists but has no measurement record
      permitted_claim_tiers:
        - interpretation
        - hypothesis
      required_action: define variables_and_measurements_needed

    observed:
      definition: behavior has been observed but not fully measured or bounded
      permitted_claim_tiers:
        - observed_behavior
      required_action: create_measurement_plan

    measured:
      definition: measurement exists with value unit method and condition
      permitted_claim_tiers:
        - measured_record
        - modeled_result
      required_action: repeat_or_bound_before_promotion

    bounded:
      definition: repeated or constrained evidence supports a limited envelope
      permitted_claim_tiers:
        - bounded_result
      required_action: review_claim_scope_before_validation

    validated:
      definition: claim approved through Governance review within explicit scope
      permitted_claim_tiers:
        - validated_claim
      required_action: preserve_scope_limits_and_anomaly_records

  promotion_rules:

    interpretation_to_hypothesis:
      allowed_when:
        - variables_identified
        - interaction_path_stated
        - unresolved_measurements_visible
      blocked_when:
        - claim_implies_validation
        - terms_are_uncontrolled
        - domain_bridge_missing

    hypothesis_to_modeled_result:
      allowed_when:
        - Engine_operator_or_format_exists
        - inputs_outputs_defined
        - assumptions_visible
        - boundary_conditions_stated
      blocked_when:
        - computation_missing_units
        - bridge_required_but_no_Interface_record
        - loss_terms_hidden

    modeled_result_to_measured_record:
      allowed_when:
        - measurement_matches_modeled_variable
        - method_and_condition_recorded
        - units_present
      blocked_when:
        - calculation_treated_as_measurement
        - observed_value_without_method
        - missing_object_identity

    measured_record_to_bounded_result:
      allowed_when:
        - repeat_count_present
        - boundary_conditions_defined
        - uncertainty_or_variance_recorded
        - anomaly_records_preserved
      blocked_when:
        - single_trial_only
        - conditions_uncontrolled
        - negative_results_removed

    bounded_result_to_validated_claim:
      allowed_when:
        - claim_text_explicit
        - tested_scope_defined
        - evidence_package_complete
        - contradictions_reviewed
        - prohibited_claims_checked
        - review_decision_recorded
      blocked_when:
        - extrapolation_beyond_tested_envelope
        - safety_or_readiness_implied_without_specific_review
        - commercial_or_operational_claim_inferred

  summary_quarantine:
    rule: >
      Summaries, polished explanations, diagrams, compressed files, or readable
      restatements are non-canonical unless routed through the owning authority lane
      and assigned an evidence or validation state.
    quarantine_triggers:
      - summary_restates_claim_without_evidence_state
      - polished_prose_implies_validation
      - diagram_or_render_implies_readiness
      - compression_hides_unresolved_variables
      - contradiction_smoothed_away
    required_repair:
      - restore_authority_pointer
      - restore_unresolved_variables
      - mark_claim_tier
      - route_to_owner_file
